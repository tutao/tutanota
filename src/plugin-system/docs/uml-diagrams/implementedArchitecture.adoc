[Proto Interface Definition] as IDL
[Custom Protobuf Plugin] as CustomGen
() "Host API" as HostAPI
() "Worker API" as WorkerAPI

CustomGen -left-> HostAPI: generates
CustomGen -right-> WorkerAPI: generates

package "Tuta Client" {
    [Host RPC Stub and Adapter] as AdapterStub
    [Business Logic] as Logic
    [Plugin Manager] as Manager
    [Plugin Loader] as Loader

    AdapterStub <-[hidden]-> Logic

    package "Web Worker" as Worker {
        [Plugin]
        [SDK]
        [Worker RPC Stub] as WorkerRPC

        SDK <-up-> WorkerRPC
        Plugin <-up-> SDK
    }

    Manager -up-> Worker: spawnes
    Loader -up[hidden]-> Plugin: loads
    Manager -[hidden]-> Loader

    AdapterStub -up-> HostAPI: implements
    WorkerRPC -up-> WorkerAPI: implements
    AdapterStub <-left-> WorkerRPC: MessagePort
}
